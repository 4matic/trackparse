1. Install the package.

    <Tabs syncKey="lang">
      <TabItem label="JavaScript">
        ```sh
        npm install trackparse   # or: pnpm add / yarn add / bun add trackparse
        ```

        Node 18 or newer, or any runtime with ES2020. ESM and CommonJS both work, and the
        TypeScript types are included. The package is
        [`trackparse` on npm](https://www.npmjs.com/package/trackparse).
      </TabItem>
      <TabItem label="Python">
        ```sh
        pip install trackparse   # or: uv add trackparse
        ```

        Python 3.9 or newer, no dependencies. The package ships `py.typed` and passes
        `mypy --strict`. The package is
        [`trackparse` on PyPI](https://pypi.org/project/trackparse/).
      </TabItem>
      <TabItem label="Rust">
        ```sh
        cargo add trackparse
        cargo add trackparse --features serde   # with Serialize and Deserialize on the types
        ```

        Rust 1.74 or newer. The only dependency is `unicode-normalization`, and there is no
        `unsafe` code. The crate is
        [`trackparse` on crates.io](https://crates.io/crates/trackparse), and its API docs are on
        [docs.rs](https://docs.rs/trackparse).
      </TabItem>
      <TabItem label="Go">
        ```sh
        go get github.com/4matic/trackparse/go@latest
        ```

        Go 1.22 or newer, no dependencies. The package name is `trackparse`. The API docs are on
        [pkg.go.dev](https://pkg.go.dev/github.com/4matic/trackparse/go).
      </TabItem>
    </Tabs>

2. Parse a track with [`parse`](/trackparse/reference/parse/)
   ([try it in the playground](/trackparse/playground/#track=Noisia%20feat.%20Foreign%20Beggars%20-%20Shellshock)).

    <Tabs syncKey="lang">
      <TabItem label="JavaScript">
        ```js
        import { parse } from "trackparse";
        // CommonJS: const { parse } = require("trackparse");

        const track = parse("Noisia feat. Foreign Beggars - Shellshock");

        track.title;                          // 'Shellshock'
        track.artists.map((a) => a.name);     // [ 'Noisia', 'Foreign Beggars' ]
        track.artists[1].role;                // 'featured'
        ```
      </TabItem>
      <TabItem label="Python">
        ```python
        from trackparse import parse

        track = parse("Noisia feat. Foreign Beggars - Shellshock")

        track.title                           # 'Shellshock'
        [a.name for a in track.artists]       # ['Noisia', 'Foreign Beggars']
        track.artists[1].role                 # 'featured'
        ```
      </TabItem>
      <TabItem label="Rust">
        ```rust
        use trackparse::parse;

        let track = parse("Noisia feat. Foreign Beggars - Shellshock");

        track.title;                                                 // "Shellshock"
        track.artists.iter().map(|a| &a.name).collect::<Vec<_>>();   // ["Noisia", "Foreign Beggars"]
        track.artists[1].role;                                       // Featured
        ```
      </TabItem>
      <TabItem label="Go">
        ```go
        import (
            "fmt"

            "github.com/4matic/trackparse/go"
        )

        track := trackparse.Parse("Noisia feat. Foreign Beggars - Shellshock")

        fmt.Printf("%q\n", track.Title)   // "Shellshock"
        names := []string{}
        for _, a := range track.Artists {
            names = append(names, a.Name)
        }
        fmt.Printf("%q\n", names)                 // ["Noisia" "Foreign Beggars"]
        fmt.Println(track.Artists[1].Role)       // featured
        ```
      </TabItem>
    </Tabs>

3. Look at the [versions](/trackparse/guides/versions/) and the
   [noise](/trackparse/guides/junk-and-flags/).

    <Tabs syncKey="lang">
      <TabItem label="JavaScript">
        ```js
        const t = parse("Daft Punk - One More Time (Official Music Video) [HD]");
        t.title;   // 'One More Time'
        t.junk;    // [ { raw: 'Official Music Video', kind: 'video' }, { raw: 'HD', kind: 'quality' } ]

        const r = parse("Dimension - Offshore (Mefjus Remix / Noisia Remix)");
        r.versions.map((v) => `${v.type} by ${v.artists[0].name}`);
        // [ 'remix by Mefjus', 'remix by Noisia' ]
        ```
      </TabItem>
      <TabItem label="Python">
        ```python
        t = parse("Daft Punk - One More Time (Official Music Video) [HD]")
        t.title   # 'One More Time'
        t.junk    # (Junk(raw='Official Music Video', kind='video'), Junk(raw='HD', kind='quality'))

        r = parse("Dimension - Offshore (Mefjus Remix / Noisia Remix)")
        [f"{v.type} by {v.artists[0].name}" for v in r.versions]
        # ['remix by Mefjus', 'remix by Noisia']
        ```
      </TabItem>
      <TabItem label="Rust">
        ```rust
        let t = parse("Daft Punk - One More Time (Official Music Video) [HD]");
        &t.title;  // "One More Time"
        &t.junk;   // [Junk { raw: "Official Music Video", kind: Video }, Junk { raw: "HD", kind: Quality }]

        let r = parse("Dimension - Offshore (Mefjus Remix / Noisia Remix)");
        r.versions.iter().map(|v| format!("{} by {}", v.version_type, v.artists[0].name)).collect::<Vec<_>>();
        // ["remix by Mefjus", "remix by Noisia"]
        ```
      </TabItem>
      <TabItem label="Go">
        ```go
        t := trackparse.Parse("Daft Punk - One More Time (Official Music Video) [HD]")
        fmt.Printf("%q\n", t.Title)   // "One More Time"
        fmt.Printf("%+v\n", t.Junk)   // [{Raw:Official Music Video Kind:video} {Raw:HD Kind:quality}]

        r := trackparse.Parse("Dimension - Offshore (Mefjus Remix / Noisia Remix)")
        for _, v := range r.Versions {
            fmt.Println(v.Type, "by", v.Artists[0].Name)
        }
        // remix by Mefjus
        // remix by Noisia
        ```
      </TabItem>
    </Tabs>

4. Tell it about names that contain a joiner with
   [`knownArtists`](/trackparse/guides/artists/#knownartists)
   ([try it](/trackparse/playground/#track=Chase%20%26%20Status%20-%20Baddadan&known=Chase%20%26%20Status)).

    <Tabs syncKey="lang">
      <TabItem label="JavaScript">
        ```js
        parse("Chase & Status - Baddadan").artists.map((a) => a.name);
        // [ 'Chase', 'Status' ]

        parse("Chase & Status - Baddadan", { knownArtists: ["Chase & Status"] }).artists.map((a) => a.name);
        // [ 'Chase & Status' ]
        ```
      </TabItem>
      <TabItem label="Python">
        ```python
        [a.name for a in parse("Chase & Status - Baddadan").artists]
        # ['Chase', 'Status']

        [a.name for a in parse("Chase & Status - Baddadan", known_artists=["Chase & Status"]).artists]
        # ['Chase & Status']
        ```
      </TabItem>
      <TabItem label="Rust">
        ```rust
        use trackparse::{parse, parse_with, ParseOptions};

        parse("Chase & Status - Baddadan").artists.iter().map(|a| &a.name).collect::<Vec<_>>();
        // ["Chase", "Status"]

        let options = ParseOptions::new().known_artists(["Chase & Status"]);
        parse_with("Chase & Status - Baddadan", &options).artists.iter().map(|a| &a.name).collect::<Vec<_>>();
        // ["Chase & Status"]
        ```
      </TabItem>
      <TabItem label="Go">
        ```go
        names = nil
        for _, a := range trackparse.Parse("Chase & Status - Baddadan").Artists {
            names = append(names, a.Name)
        }
        fmt.Printf("%q\n", names)   // ["Chase" "Status"]

        options := trackparse.ParseOptions{KnownArtists: []string{"Chase & Status"}}
        names = nil
        for _, a := range trackparse.ParseWith("Chase & Status - Baddadan", options).Artists {
            names = append(names, a.Name)
        }
        fmt.Printf("%q\n", names)   // ["Chase & Status"]
        ```
      </TabItem>
    </Tabs>

5. Write it back with [`format`](/trackparse/guides/formatting/), or turn it into
   [JSON](/trackparse/reference/parse/#serializing-to-json).

    <Tabs syncKey="lang">
      <TabItem label="JavaScript">
        ```js
        import { format } from "trackparse";

        format(t);           // 'Daft Punk - One More Time'
        JSON.stringify(t);   // '{"input":"Daft Punk - One More Time (Official Music Video) [HD]","mode":"youtube",…}'
        ```
      </TabItem>
      <TabItem label="Python">
        ```python
        import json
        from trackparse import format_track

        format_track(t)            # 'Daft Punk - One More Time'
        json.dumps(t.to_dict())    # '{"input": "Daft Punk - One More Time (Official Music Video) [HD]", "mode": "youtube", …}'
        ```
      </TabItem>
      <TabItem label="Rust">
        ```rust
        use trackparse::format_track;

        format_track(&t);   // "Daft Punk - One More Time"
        t.to_json();        // "{\"input\":\"Daft Punk - One More Time (Official Music Video) [HD]\",\"mode\":\"youtube\",…}"
        ```
      </TabItem>
      <TabItem label="Go">
        ```go
        import "encoding/json"

        fmt.Println(trackparse.Format(t))   // Daft Punk - One More Time
        data, _ := json.Marshal(t)
        fmt.Println(string(data))           // {"input":"Daft Punk - One More Time (Official Music Video) [HD]","mode":"youtube",…}
        ```
      </TabItem>
    </Tabs>

6. Parse a whole [tracklist](/trackparse/guides/tracklists/).

    <Tabs syncKey="lang">
      <TabItem label="JavaScript">
        ```js
        import { parseTracklist } from "trackparse/tracklist";

        const list = parseTracklist("00:00 Noisia - Stigma\n04:10 Mefjus - Sleepless (Phace Remix)");
        list.entries.map((e) => `${e.time.seconds}s ${e.track.title}`);   // [ '0s Stigma', '250s Sleepless' ]
        list.warnings;                                                    // []
        ```
      </TabItem>
      <TabItem label="Python">
        ```python
        from trackparse.tracklist import parse_tracklist

        tl = parse_tracklist("00:00 Noisia - Stigma\n04:10 Mefjus - Sleepless (Phace Remix)")
        [f"{e.time.seconds}s {e.track.title}" for e in tl.entries]   # ['0s Stigma', '250s Sleepless']
        tl.warnings                                                  # ()
        ```
      </TabItem>
      <TabItem label="Rust">
        ```rust
        use trackparse::tracklist::parse_tracklist;

        let list = parse_tracklist("00:00 Noisia - Stigma\n04:10 Mefjus - Sleepless (Phace Remix)");
        list.entries.iter()
            .filter_map(|e| Some(format!("{}s {}", e.time.as_ref()?.seconds, e.track.as_ref()?.title)))
            .collect::<Vec<_>>();   // ["0s Stigma", "250s Sleepless"]
        list.warnings;              // []
        ```

        `time` and `track` are `Option`s in Rust, because a blank line or a note has neither.
      </TabItem>
      <TabItem label="Go">
        ```go
        list := trackparse.ParseTracklist("00:00 Noisia - Stigma\n04:10 Mefjus - Sleepless (Phace Remix)")
        for _, e := range list.Entries {
            if e.Time != nil && e.Track != nil {
                fmt.Printf("%vs %s\n", e.Time.Seconds, e.Track.Title)
            }
        }
        // 0s Stigma
        // 250s Sleepless
        fmt.Println(len(list.Warnings))   // 0
        ```

        `Time` and `Track` are pointers in Go, `nil` on a blank line or a note.
      </TabItem>
    </Tabs>

</Steps>

<Aside type="tip" title="Parsing a lot of strings?">
`createParser(options)` ([`create_parser`](/trackparse/reference/functions/#createparseroptions) in Python, `Parser::new(&options)` in Rust,
`trackparse.NewParser(options)` in Go) compiles your options and keywords once. Its [`parse`](/trackparse/reference/parse/) method is the same function with those options already applied. See [Other functions](/trackparse/reference/functions/#createparseroptions).
</Aside>
