## Extract, don't delete

Most title parsers split on the dash and hand you two strings. Others "clean" the title and throw the
featured artist and the remix away with the brackets.

trackparse keeps everything. What it recognizes goes into a labeled field, and what it doesn't stays
in the title, word for word.

## Same call, same answer

<Tabs syncKey="lang">
  <TabItem label="JavaScript">
    ```js
    import { parse } from "trackparse";

    const track = parse("Noisia feat. Foreign Beggars - Shellshock");
    track.artists.map((a) => `${a.name} (${a.role})`);
    // [ 'Noisia (primary)', 'Foreign Beggars (featured)' ]

    parse("Queen - Bohemian Rhapsody (Live at Wembley 1986)").versions[0];
    // { type: 'live', descriptor: 'at Wembley', year: 1986, … }
    ```
  </TabItem>
  <TabItem label="Python">
    ```python
    from trackparse import parse

    track = parse("Noisia feat. Foreign Beggars - Shellshock")
    [f"{a.name} ({a.role})" for a in track.artists]
    # ['Noisia (primary)', 'Foreign Beggars (featured)']

    parse("Queen - Bohemian Rhapsody (Live at Wembley 1986)").versions[0]
    # Version(type='live', ..., descriptor='at Wembley', year=1986, ...)
    ```
  </TabItem>
  <TabItem label="Rust">
    ```rust
    use trackparse::parse;

    let track = parse("Noisia feat. Foreign Beggars - Shellshock");
    track.artists.iter().map(|a| format!("{} ({})", a.name, a.role)).collect::<Vec<_>>();
    // ["Noisia (primary)", "Foreign Beggars (featured)"]

    parse("Queen - Bohemian Rhapsody (Live at Wembley 1986)").versions.first();
    // Some(Version { version_type: Live, descriptor: Some("at Wembley"), year: Some(1986), .. })
    ```
  </TabItem>
  <TabItem label="Go">
    ```go
    import (
        "fmt"

        "github.com/4matic/trackparse/go"
    )

    track := trackparse.Parse("Noisia feat. Foreign Beggars - Shellshock")
    for _, a := range track.Artists {
        fmt.Printf("%s (%s)\n", a.Name, a.Role)
    }
    // Noisia (primary)
    // Foreign Beggars (featured)

    v := trackparse.Parse("Queen - Bohemian Rhapsody (Live at Wembley 1986)").Versions[0]
    fmt.Println(v.Type, *v.Descriptor, *v.Year)
    // live at Wembley 1986
    ```
  </TabItem>
</Tabs>
